#!/bin/sh
# Opt-in, seconds-only local checks. Git supplies pushed local OIDs on stdin.
# CI remains authoritative; Git's --no-verify bypasses this hook.
set -eu
cd "$(git rev-parse --show-toplevel)"
repo=$(pwd)
snapshot=$(mktemp -d "${TMPDIR:-/tmp}/picsee-pre-push.XXXXXXXX")
trap 'rm -rf "$snapshot"' 0
trap 'exit 1' HUP INT TERM

saw_input=0
while read -r local_ref local_oid remote_ref remote_oid; do
  saw_input=1
  # A deletion has no commit/body to validate (also supports SHA-256 zero OIDs).
  case "$local_oid" in
    '') echo 'pre-push: missing pushed local commit' >&2; exit 2 ;;
    *[!0]*) ;;
    *) continue ;;
  esac
  # Archive each pushed commit, including non-HEAD refs. Separate archive/extract
  # commands fail closed without a pipeline masking git failures. Each ref gets a
  # fresh directory so files deleted by the next commit cannot leak across checks.
  git archive --format=tar --output="$snapshot/input.tar" "$local_oid"
  mkdir "$snapshot/tree"
  tar -xf "$snapshot/input.tar" -C "$snapshot/tree"
  (
    cd "$snapshot/tree"
    node scripts/ledger/build.mjs --check
    node scripts/design/shortcut-model.mjs
  )
  rm -rf "$snapshot/tree"
  if [ -n "${PICSEE_PR_BODY:-}" ]; then
    node "$repo/scripts/agents/local-pr-lint.mjs" --body-file "$PICSEE_PR_BODY" --expect "$local_oid"
  fi
done
if [ -n "${PICSEE_PR_BODY:-}" ] && [ "$saw_input" = 0 ]; then
  echo 'pre-push: no push input; cannot bind PR body to the pushed commit' >&2
  exit 2
fi
